import java.io.*;
import java.util.*;

public class LRU {
    public static void main(String[] args) throws IOException {

        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));

        // Input variables
        int totalFrames;             // Number of available frames in memory
        int pointer = 0;             // Keeps track of which frame to replace next
        int hitCount = 0;            // Number of page hits
        int faultCount = 0;          // Number of page faults
        int referenceLength;         // Length of the reference string
        boolean isMemoryFull = false;

        // Data structures
        int[] frameBuffer;           // Stores current pages in memory
        int[] referenceString;       // Stores all pages to be referenced
        int[][] memoryLayout;        // Stores memory state after each reference
        ArrayList<Integer> recentPages = new ArrayList<>(); // Keeps pages in order of use (for LRU)

        // Input from user
        System.out.print("Enter number of Frames: ");
        totalFrames = Integer.parseInt(reader.readLine());

        System.out.print("Enter length of Reference String: ");
        referenceLength = Integer.parseInt(reader.readLine());

        referenceString = new int[referenceLength];
        memoryLayout = new int[referenceLength][totalFrames];
        frameBuffer = new int[totalFrames];

        // Initialize all frames as empty (-1)
        for (int i = 0; i < totalFrames; i++)
            frameBuffer[i] = -1;

        System.out.println("Enter the Reference String: ");
        for (int i = 0; i < referenceLength; i++) {
            referenceString[i] = Integer.parseInt(reader.readLine());
        }

        System.out.println();

        // Start processing each page in the reference string
        for (int i = 0; i < referenceLength; i++) {

            // If page is already in stack (recently used list), remove it
            if (recentPages.contains(referenceString[i])) {
                recentPages.remove(recentPages.indexOf(referenceString[i]));
            }

            // Add current page to the top (most recently used)
            recentPages.add(referenceString[i]);

            int pageFoundAt = -1;

            // Check if page is already in one of the frames (HIT)
            for (int j = 0; j < totalFrames; j++) {
                if (frameBuffer[j] == referenceString[i]) {
                    pageFoundAt = j;
                    hitCount++;
                    break;
                }
            }

            // If page not found ? Page Fault occurs
            if (pageFoundAt == -1) {
                // If memory already full, find least recently used page to replace
                if (isMemoryFull) {
                    int leastRecentlyUsedIndex = referenceLength; // large initial value
                    for (int j = 0; j < totalFrames; j++) {
                        if (recentPages.contains(frameBuffer[j])) {
                            int indexInStack = recentPages.indexOf(frameBuffer[j]);
                            if (indexInStack < leastRecentlyUsedIndex) {
                                leastRecentlyUsedIndex = indexInStack;
                                pointer = j; // frame to replace
                            }
                        }
                    }
                }

                // Replace the selected frame with new page
                frameBuffer[pointer] = referenceString[i];
                faultCount++;
                pointer++;

                // If pointer exceeds frame count, reset and mark memory as full
                if (pointer == totalFrames) {
                    pointer = 0;
                    isMemoryFull = true;
                }
            }

            // Store memory layout for this step
            for (int j = 0; j < totalFrames; j++)
                memoryLayout[i][j] = frameBuffer[j];
        }

        // Display memory layout (each row = frame, each column = step)
        System.out.println("\nMemory Layout:");
        for (int i = 0; i < totalFrames; i++) {
            for (int j = 0; j < referenceLength; j++) {
                System.out.printf("%3d ", memoryLayout[j][i]);
            }
            System.out.println();
        }

        // Print final statistics
        System.out.println("\nTotal Page Hits: " + hitCount);
        System.out.println("Total Page Faults: " + faultCount);
        System.out.println("Hit Ratio: " + (float) hitCount / referenceLength);
    }
}
