import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;

public class OptimalReplacement {
    public static void main(String[] args) throws IOException {

        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

        System.out.print("Enter the number of frames: ");
        int totalFrames = Integer.parseInt(br.readLine());

        System.out.print("Enter the length of the reference string: ");
        int refLength = Integer.parseInt(br.readLine());

        int[] referenceString = new int[refLength];   // To store the reference pages
        int[] memory = new int[totalFrames];          // To represent frames in memory
        int[][] memoryLayout = new int[refLength][totalFrames]; // For display purpose

        // Initialize memory with -1 (empty)
        for (int i = 0; i < totalFrames; i++)
            memory[i] = -1;

        // Take reference string input
        System.out.println("Enter the reference string:");
        for (int i = 0; i < refLength; i++)
            referenceString[i] = Integer.parseInt(br.readLine());

        int pointer = 0;          // Points to the next frame to replace
        int hitCount = 0;         // Number of hits
        int faultCount = 0;       // Number of page faults
        boolean isMemoryFull = false;

        for (int i = 0; i < refLength; i++) {
            int currentPage = referenceString[i];
            boolean isHit = false;

            //  Check if current page is already in memory
            for (int j = 0; j < totalFrames; j++) {
                if (memory[j] == currentPage) {
                    hitCount++;
                    isHit = true;
                    break;
                }
            }

            // If page is not found ? Page Fault occurs
            if (!isHit) {
                // If memory is full, apply Optimal Replacement Logic
                if (isMemoryFull) {
                    int[] nextUseIndex = new int[totalFrames];      // To store next use position
                    boolean[] foundLater = new boolean[totalFrames];// To mark if page is found later

                    // For each page currently in memory, find when it will be used next
                    for (int j = i + 1; j < refLength; j++) {
                        for (int k = 0; k < totalFrames; k++) {
                            if (referenceString[j] == memory[k] && !foundLater[k]) {
                                nextUseIndex[k] = j;
                                foundLater[k] = true;
                                break;
                            }
                        }
                    }

                    // Replace the page that will not be used for the longest time
                    int farthest = -1;
                    int replaceIndex = -1;

                    for (int j = 0; j < totalFrames; j++) {
                        // If a page is not used again, replace it immediately
                        if (!foundLater[j]) {
                            replaceIndex = j;
                            break;
                        }
                        if (nextUseIndex[j] > farthest) {
                            farthest = nextUseIndex[j];
                            replaceIndex = j;
                        }
                    }

                    // Replace page
                    pointer = replaceIndex;
                }

                // Put the new page into memory
                memory[pointer] = currentPage;
                faultCount++;

                // When frames are not yet full, move the pointer sequentially
                if (!isMemoryFull) {
                    pointer++;
                    if (pointer == totalFrames) {
                        pointer = 0;
                        isMemoryFull = true;
                    }
                }
            }

            // Store the current memory layout for display
            for (int j = 0; j < totalFrames; j++)
                memoryLayout[i][j] = memory[j];
        }

        System.out.println("\nMemory Layout (Each column = one step):");
        for (int i = 0; i < totalFrames; i++) {
            for (int j = 0; j < refLength; j++) {
                System.out.printf("%3d ", memoryLayout[j][i]);
            }
            System.out.println();
        }
        System.out.println("\nTotal Hits: " + hitCount);
        System.out.println("Total Faults: " + faultCount);
        System.out.println("Hit Ratio: " + (float) hitCount / refLength);
    }
}
